     HOW PASSWORD WORKS
 User creates password: Sunset!2024
System generates a random salt: x9Kp2m
System combines them: x9Kp2mSunset!2024
System hashes it (one-way math):
hash("x9Kp2mSunset!2024") → "a1f8c9d4e7b2..."
System stores ONLY:
salt: x9Kp2m | hash: a1f8c9d4e7b2...


     CODE EXAMPLE
     import hashlib, secrets

# 1. Creating a new account
password = "Sunset!2024"
salt = secrets.token_hex(8)  # e.g., "a3f9c1b2"
stored_hash = hashlib.sha256((salt + password).encode()).hexdigest()
print(f"Stored: salt={salt} | hash={stored_hash}")

# 2. Later, during login
login_attempt = "Sunset!2024"
attempt_hash = hashlib.sha256((salt + login_attempt).encode()).hexdigest()
print("✅ Match" if attempt_hash == stored_hash else "❌ Wrong")

      DEHASHING

      import hashlib
import itertools
import string
import time

def crack_hash_simulated(target_hash: str, algorithm: str = "sha256", max_len: int = 4):
    """
    Educational simulation of hash cracking via brute force.
    WARNING: Only works on extremely short passwords. Real systems use salts + slow hashes.
    """
    chars = string.ascii_lowercase + string.digits  # a-z, 0-9
    attempt_count = 0

    print(f"🎯 Target: {target_hash}")
    print("⏳ Trying combinations...")

    # Try lengths 1 to max_len
    for length in range(1, max_len + 1):
        for guess_tuple in itertools.product(chars, repeat=length):
            guess = "".join(guess_tuple)
            attempt_count += 1

            # Hash the guess
            h = hashlib.new(algorithm)
            h.update(guess.encode())
            guess_hash = h.hexdigest()

            # Compare
            if guess_hash == target_hash:
                print(f"✅ MATCH FOUND: '{guess}'")
                print(f"📊 Attempts: {attempt_count}")
                return guess

            # Simulate rate limiting (0.001s = 1,000 guesses/sec max)
            time.sleep(0.001)

            if attempt_count % 1000 == 0:
                print(f"   🔍 Tried {attempt_count} so far...")

    print("❌ No match found within length limit.")
    return None



# 🔐 Example Usage (EducationaL)
if __name__ == "__main__":
    print("⚠️  This is a local simulation. Never target systems without explicit permission.\n")

    # Create a known weak hash
    TARGET = hashlib.sha256(b"cat1").hexdigest()
    crack_hash_simulated(TARGET, max_len=4)
